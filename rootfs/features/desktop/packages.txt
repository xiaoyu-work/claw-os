# desktop — claw-os desktop environment runtime + build dependencies.
#
# The desktop is built from source by install.sh. The packages below are
# the union of:
#   - build deps for the Rust + Wayland desktop tree
#   - runtime libs needed to actually run the compositor + shell
#   - pam / polkit / seatd needed by the display manager and session

build-essential
pkg-config
clang
mold
lld
just
git

libwayland-dev
libwayland-client0
libwayland-server0
