# desktop — claw-os desktop environment build + runtime dependencies.
#
# Target distro: Debian 13 "trixie" (kernel 6.12 LTS, PipeWire 1.4, Mesa 24).
#
# The desktop is built from the vendored monorepo under <repo>/desktop/.
# install.sh runs `just build` + `just install` against that tree to put the
# binaries / .desktop / systemd unit files into the rootfs, then wires up
# the greeter / display manager / user services so the system boots into a
# graphical login.

# pam_gnome_keyring.so lives in this separate package; without it the
# `-session optional pam_gnome_keyring.so auto_start` line in
# /etc/pam.d/greetd silently no-ops (the `-` prefix means "ignore if
# the module is missing"). When that happens the keyring daemon is
# never primed with the user's password at login, so libsecret-using
# apps (Chromium, GNOME apps, etc.) prompt for a master password on
# first launch.
libpam-gnome-keyring

# --- Fonts and icons ---

# --- Default user-facing GUI apps ---
# Chromium is the default GUI web browser. It is also the engine the
# `claw-agent-browser` MV3 extension + apps/browser-attached bridge
# attach to when the AI agent needs the user's logged-in session
# (Browser Mode B). Bound to x-scheme-handler/{http,https} and
# text/html in overlay/etc/xdg/mimeapps.list. The headless cos-browser
# CLI (`rootfs/features/browser`) is a separate, AI-only surface and
# does not provide a desktop launcher.
chromium

# Thunderbird is the default mail / news / calendar client. The AI
# Agent's apps/email continues to drive Gmail/Outlook via their HTTPS
# APIs; Thunderbird is the GUI side users open to see their inbox.
# Bound to x-scheme-handler/mailto in overlay/etc/xdg/mimeapps.list.
thunderbird

# --- Local document full-text search (apps/docs) ---
# Recoll provides the Xapian-backed full-text index and the recollq /
# recollindex / recoll CLIs. apps/docs wraps recollq + recollindex to
# expose docs.search / docs.index / docs.status / docs.configure to the
# AI agent. Defaults to indexing ~/Documents ~/Desktop ~/Downloads
# (see overlay/etc/skel/.recoll/recoll.conf for the seeded user config).
# Real-time monitoring is enabled out of the box via the systemd --user
# unit claw-recoll-index.service (overlay/etc/systemd/user/), which runs
# `recollindex -m` and uses inotify to keep the index current on every
# create / modify / delete.
# Pulls in poppler-utils, antiword, unrtf, etc. as recommends for the
# PDF / Office / RTF format filters.
recoll
